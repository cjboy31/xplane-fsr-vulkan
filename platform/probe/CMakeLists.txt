cmake_minimum_required(VERSION 3.20)
project(XPFSRNativeProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(XPLANE_SDK "" CACHE PATH "Official X-Plane SDK root")
if(NOT EXISTS "${XPLANE_SDK}/CHeaders/XPLM/XPLMPlugin.h")
  message(FATAL_ERROR "Set XPLANE_SDK to the extracted official SDK")
endif()
add_library(native_probe MODULE probe.cpp)
target_include_directories(native_probe PRIVATE "${XPLANE_SDK}/CHeaders/XPLM")
target_compile_definitions(native_probe PRIVATE XPLM200=1 XPLM210=1 XPLM300=1)
target_compile_options(native_probe PRIVATE -Wall -Wextra -Werror)
set_target_properties(native_probe PROPERTIES PREFIX "" SUFFIX ".xpl" CXX_VISIBILITY_PRESET hidden)
if(APPLE)
  target_compile_definitions(native_probe PRIVATE APL=1 IBM=0 LIN=0)
  target_link_options(native_probe PRIVATE "-undefined" "dynamic_lookup")
  set_target_properties(native_probe PROPERTIES OUTPUT_NAME mac)
elseif(UNIX)
  target_compile_definitions(native_probe PRIVATE APL=0 IBM=0 LIN=1)
  set_target_properties(native_probe PROPERTIES OUTPUT_NAME lin)
else()
  message(FATAL_ERROR "Native preview targets Linux/macOS; regular Windows builds use build.ps1")
endif()
add_executable(probe_test ../tests/probe_test.cpp)
target_link_libraries(probe_test PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(probe_test PRIVATE -Wall -Wextra -Werror)
target_compile_options(probe_test PRIVATE -UNDEBUG)
target_link_options(probe_test PRIVATE -rdynamic)
enable_testing()
add_test(NAME native_probe_lifecycle COMMAND probe_test $<TARGET_FILE:native_probe>)
